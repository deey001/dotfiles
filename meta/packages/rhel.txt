# ==============================================================================
# meta/packages/rhel.txt — System Packages for RHEL / Fedora / CentOS
# ==============================================================================
# Used by: scripts/install.sh (via install_package_list)
# Command: sudo dnf install -y <packages>  (or yum on older RHEL/CentOS)
#
# Notes:
#   - Some packages require EPEL: sudo dnf install -y epel-release
#   - 'bat' may not be in default repos on RHEL 8 — install from GitHub releases
#     or enable COPR: dnf copr enable atim/bat
#   - 'fastfetch' may require RPM Fusion or COPR
#   - On RHEL/CentOS 8, some packages may use different names (fd-find vs fd)
# Format: one package name per line; lines starting with # are comments.
# ==============================================================================

# ── Build Essentials ──────────────────────────────────────────────────────────
git                    # Version control — required by stow, fzf-tab auto-clone, etc.
gcc                    # C compiler
gcc-c++                # C++ compiler
make                   # GNU Make — build automation
gawk                   # GNU AWK — used in .bashrc PATH deduplication
unzip                  # .zip extraction

# ── Core CLI Tools ───────────────────────────────────────────────────────────
tmux                   # Terminal multiplexer — persistent sessions, split panes
fzf                    # Fuzzy finder — Ctrl+R history, Ctrl+T file picker, Alt+C cd
neovim                 # Editor — LazyVim config lives in stow/config/.config/nvim/
hstr                   # History browser — alternative Ctrl+R
bat                    # cat replacement — syntax highlighting, Catppuccin theme (needs EPEL/COPR)
fastfetch              # System info display on login (may need COPR)
cmatrix                # Matrix rain screensaver
btop                   # Resource monitor — replaces top/htop (aliased as 'top')
lazygit                # TUI git client (may need COPR: atim/lazygit)
glow                   # Markdown renderer — render .md files in terminal
tldr                   # Simplified man pages
ripgrep                # grep replacement — faster, respects .gitignore, Rust-based
fd-find                # find replacement (note: Fedora uses 'fd-find', binary is 'fdfind')

# ── Shell & Completion ────────────────────────────────────────────────────────
zsh                    # Z shell — alternative to bash
zsh-syntax-highlighting  # Colorize zsh commands as you type
zsh-autosuggestions    # Fish-style ghost-text suggestions from zsh history
